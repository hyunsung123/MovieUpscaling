cmake_minimum_required(VERSION 3.24)
project(WindowGPUPreview VERSION 0.3.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Windows with MSVC and Windows SDK 10.0.19041+ is required (C++/WinRT).")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(preview_core STATIC
  src/capture/WindowCapture.cpp src/renderer/D3DRenderer.cpp src/renderer/GPUUpscaler.cpp
  src/renderer/RCASPass.cpp src/renderer/GPUCompositor.cpp
  src/utils/WindowList.cpp src/utils/GPUInfo.cpp)
target_include_directories(preview_core PUBLIC src)
target_compile_definitions(preview_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(preview_core PUBLIC /W4 /permissive- /EHsc /utf-8)
target_link_libraries(preview_core PUBLIC d3d11 dxgi d3dcompiler windowsapp dwmapi user32 gdi32 comctl32)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/shaders/present.hlsl" PRESENT_SHADER)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/shaders/common.hlsl" COMMON_SHADER)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/shaders/upscale_bicubic.hlsl" BICUBIC_SHADER)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/shaders/upscale_lanczos.hlsl" LANCZOS_SHADER)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/shaders/upscale_easu.hlsl" EASU_SHADER)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/shaders/upscale_bilinear.hlsl" BILINEAR_SHADER)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/shaders/rcas.hlsl" RCAS_SHADER)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/shaders/composite.hlsl" COMPOSITE_SHADER)
configure_file(src/shaders/ShaderSource.h.in generated/ShaderSource.h @ONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS src/shaders/present.hlsl
  src/shaders/common.hlsl src/shaders/upscale_bicubic.hlsl src/shaders/upscale_lanczos.hlsl src/shaders/upscale_easu.hlsl
  src/shaders/upscale_bilinear.hlsl src/shaders/rcas.hlsl src/shaders/composite.hlsl)
target_include_directories(preview_core PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
add_executable(WindowGPUPreview WIN32 src/main.cpp src/ui/AppUI.cpp)
target_link_libraries(WindowGPUPreview PRIVATE preview_core)
add_custom_command(TARGET WindowGPUPreview POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
  "${CMAKE_CURRENT_SOURCE_DIR}/THIRD_PARTY_NOTICES.md" "$<TARGET_FILE_DIR:WindowGPUPreview>/THIRD_PARTY_NOTICES.md")
target_link_options(WindowGPUPreview PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/app.manifest)
set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT WindowGPUPreview)
include(CTest)
if(BUILD_TESTING)
  add_executable(CaptureSmokeTest tests/CaptureSmokeTest.cpp)
  target_link_libraries(CaptureSmokeTest PRIVATE preview_core)
  add_test(NAME capture_smoke COMMAND CaptureSmokeTest)
  set_tests_properties(capture_smoke PROPERTIES TIMEOUT 30)
  add_executable(AppSmokeTest tests/AppSmokeTest.cpp)
  target_link_libraries(AppSmokeTest PRIVATE preview_core)
  add_test(NAME app_smoke COMMAND AppSmokeTest $<TARGET_FILE:WindowGPUPreview>)
  set_tests_properties(app_smoke PROPERTIES TIMEOUT 30)
  add_executable(UpscaleGPUTest tests/UpscaleGPUTest.cpp)
  target_link_libraries(UpscaleGPUTest PRIVATE preview_core)
  add_test(NAME upscale_gpu COMMAND UpscaleGPUTest)
  set_tests_properties(upscale_gpu PROPERTIES TIMEOUT 60)
  add_test(NAME upscale_capture_ui COMMAND AppSmokeTest $<TARGET_FILE:WindowGPUPreview> --phase4)
  set_tests_properties(upscale_capture_ui PROPERTIES TIMEOUT 45)
  set_tests_properties(capture_smoke app_smoke upscale_gpu upscale_capture_ui PROPERTIES RUN_SERIAL TRUE)
  add_executable(Phase5GPUTest tests/Phase5GPUTest.cpp)
  target_link_libraries(Phase5GPUTest PRIVATE preview_core)
  add_test(NAME phase5_gpu COMMAND Phase5GPUTest)
  set_tests_properties(phase5_gpu PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE)
  add_test(NAME phase5_capture_ui COMMAND AppSmokeTest $<TARGET_FILE:WindowGPUPreview> --phase5)
  set_tests_properties(phase5_capture_ui PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE)
endif()
